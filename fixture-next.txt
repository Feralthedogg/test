"""Contract tests for the independently authored pricing fixture."""

from pathlib import Path
import runpy
import unittest


class TotalCentsTests(unittest.TestCase):
    @classmethod
    def setUpClass(cls):
        implementation = runpy.run_path(Path(__file__).with_name("fixture.txt"))
        cls.total_cents = staticmethod(implementation["total_cents"])

    def test_undiscounted_total(self):
        self.assertEqual(self.total_cents(1_001, 3), 3_003)

    def test_fractional_discount(self):
        self.assertEqual(self.total_cents(1_001, 3, 1_250), 2_628)

    def test_zero_quantity(self):
        self.assertEqual(self.total_cents(1_001, 0, 1_250), 0)

    def test_zero_unit_price(self):
        self.assertEqual(self.total_cents(0, 3, 1_250), 0)

    def test_full_discount(self):
        self.assertEqual(self.total_cents(1_001, 3, 10_000), 0)

    def test_half_up_rounding_boundary(self):
        for discount_bps, expected in ((4_999, 1), (5_000, 1), (5_001, 0)):
            with self.subTest(discount_bps=discount_bps):
                self.assertEqual(self.total_cents(1, 1, discount_bps), expected)

    def test_rounding_applies_to_the_total(self):
        self.assertEqual(self.total_cents(1, 3, 5_000), 2)

    def test_large_integer_precision(self):
        self.assertEqual(
            self.total_cents(9_007_199_254_740_993, 3, 1_250),
            23_643_898_043_695_107,
        )

    def test_boolean_arguments_are_rejected(self):
        for position in range(3):
            for value in (False, True):
                arguments = [100, 2, 0]
                arguments[position] = value
                with self.subTest(position=position, value=value):
                    with self.assertRaises(TypeError):
                        self.total_cents(*arguments)

    def test_other_noninteger_arguments_are_rejected(self):
        for position in range(3):
            for value in (1.5, "1", None, (), [], {}):
                arguments = [100, 2, 0]
                arguments[position] = value
                with self.subTest(position=position, value=value):
                    with self.assertRaises(TypeError):
                        self.total_cents(*arguments)

    def test_negative_prices_and_quantities_are_rejected(self):
        for unit_cents, quantity in ((-1, 2), (100, -1)):
            with self.subTest(unit_cents=unit_cents, quantity=quantity):
                with self.assertRaises(ValueError):
                    self.total_cents(unit_cents, quantity)

    def test_out_of_range_discounts_are_rejected(self):
        for discount_bps in (-1, 10_001):
            with self.subTest(discount_bps=discount_bps):
                with self.assertRaises(ValueError):
                    self.total_cents(100, 2, discount_bps)


if __name__ == "__main__":
    unittest.main(verbosity=2)
